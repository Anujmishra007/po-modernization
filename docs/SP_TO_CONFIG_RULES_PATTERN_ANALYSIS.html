Pattern Classification

Pattern: Configuration Lookup
Legacy T-SQL
SELECT @value = configvalue
FROM storerconfig
WHERE storerkey = @StorerKey
  AND configkey = 'AllowOverReceipt'
Modern Java
boolean allowOverReceipt =
  validationConfig
    .allowOverReceipt(storerKey);
Pattern: Validation with Threshold
Legacy T-SQL
IF @ReceivedQty > @ExpectedQty
   * (1 + @Tolerance/100)
  RAISERROR('Over-receipt
    exceeds tolerance', 16, 1)
Modern Java
rule "Over-receipt validation"
  when
    $line : ReceiptLine(
      qtyReceived > qtyExpected
        * (1 + tolerance))
  then
    $line.addError("...");
Pattern: Client-Specific Branch
Legacy T-SQL
IF @StorerKey = 'NIKE'
  EXEC ispNikeLogic @Key
ELSE IF @StorerKey = 'HM'
  EXEC ispHMLogic @Key
ELSE IF... (50+ more)
Modern Java
List plugins =
  pluginRegistry.getPlugins(ctx);

for (var plugin : plugins) {
  plugin.execute(receipt, ctx);
}
Pattern: Lottable Derivation
Legacy T-SQL
IF @Lottable02 IS NULL
   AND @SupplierLot IS NOT NULL
  SET @Lottable02 = @SupplierLot
ELSE IF @Lottable02 IS NULL
  SET @Lottable02 = 'RCV-'
    + @ReceiptKey + '-' + @Line
Modern Java
rule "Lottable02 from supplier"
  salience 100
  when
    $d : ReceiptDetailFact(
      lottable02 == null,
      supplierLot != null)
  then
    $d.setLottable02(
      $d.getSupplierLot());
Pattern: Status Transition
Legacy T-SQL
IF @CurrentStatus NOT IN ('0', '5')
  RAISERROR('Invalid status
    for finalization', 16, 1)
UPDATE RECEIPT
SET Status = '9'
WHERE ReceiptKey = @Key
Modern Java
if (!statusConfig.isValidTransition(
    "RECEIPT", currentStatus, "9"))
  throw new BusinessException(
    "Invalid status transition");

receipt.setStatus(
  StatusConfig.RECEIVED);
